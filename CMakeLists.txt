cmake_minimum_required(VERSION 3.25)
project(NPCsHaveAmmo VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/CommonLibSSE-NG)
include(vendor/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(NPCsHaveAmmo NAME "NPCs Have Ammo (arrow/bolt restock mod)" AUTHOR "Maxmc" VERSION 1.0.3 SOURCES src/Plugin.cpp)
target_include_directories(NPCsHaveAmmo PRIVATE src)
target_precompile_headers(NPCsHaveAmmo PRIVATE src/PCH.h)
target_compile_definitions(NPCsHaveAmmo PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NPCsHaveAmmo PRIVATE /W4 /permissive-)
set_target_properties(NPCsHaveAmmo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
enable_testing()
add_executable(AmmoTests tests/AmmoTests.cpp)
target_include_directories(AmmoTests PRIVATE src)
add_test(NAME AmmoBehavior COMMAND AmmoTests)
